export function effortFloor(text: string): number {

}

핵심은 하한이 무성의에만 붙는다는 점입니다. 무례·조롱은 none으로 분류되지 않으니 DELTA_MIN까지 그대로 통과합니다 — 여기까지 막으면 감점 장치 자체가 죽습니다. 테스트로 양쪽을 다 고정했습니다(96개 통과, +12).
